# Met à jour les cours et les titres de MacroViewer automatiquement
name: Mise à jour des cours

on:
  schedule:
    - cron: "*/15 6-21 * * 1-5"   # toutes les 15 min en semaine, de 7 h à 23 h (heure de Paris)
    - cron: "0 */4 * * 0,6"       # toutes les 4 h le week-end (devises, titres de presse)
  workflow_dispatch:              # bouton « Run workflow » pour lancer une mise à jour à la main

permissions:
  contents: write

concurrency:
  group: mise-a-jour
  cancel-in-progress: false

jobs:
  update:
    runs-on: windows-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4

      - name: Relever les cours et les titres
        shell: powershell
        env:
          # cours de référence des concours : envoyés au serveur Supabase avec la clé secrète
          SUPABASE_URL: https://kuzqmyvswslbucuwjczx.supabase.co
          SUPABASE_SERVICE_KEY: ${{ secrets.SUPABASE_SERVICE_KEY }}
        run: ./update.ps1 -OutDir .

      - name: Publier les nouveaux fichiers
        shell: bash
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898283+github-actions[bot]@users.noreply.github.com"
          git add quotes.json news.json
          if git diff --cached --quiet; then echo "Rien de nouveau"; exit 0; fi
          git commit -m "Mise à jour des cours"
          git pull --rebase
          git push
